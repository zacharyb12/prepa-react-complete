# useReducer

### Qu'est-ce que useReducer ?
`useReducer` est un hook qui permet de gérer un état complexe de manière plus prévisible qu'avec `useState`. Il utilise le pattern Redux avec des actions et un reducer.

### Quand utiliser useReducer ?
- État complexe avec plusieurs propriétés liées
- Logique de mise à jour complexe
- Plusieurs actions qui modifient l'état
- Quand useState devient difficile à gérer

---

## Syntaxe de base

```tsx
import { useReducer } from 'react';

const [state, dispatch] = useReducer(reducer, initialState);
```

- `state` : l'état actuel
- `dispatch` : fonction pour déclencher des actions
- `reducer` : fonction qui détermine comment l'état change
- `initialState` : état initial

---

## Exemple simple : Compteur

### 1. Définir les types et l'état initial

```tsx
// Types pour les actions
type CounterAction = 
  | { type: 'INCREMENT' }
  | { type: 'DECREMENT' }
  | { type: 'RESET' }
  | { type: 'SET_VALUE'; payload: number };

// État initial
const initialState = {
  count: 0
};

type CounterState = typeof initialState;
```

### 2. Créer le reducer

```tsx
function counterReducer(state: CounterState, action: CounterAction): CounterState {
  switch (action.type) {
    case 'INCREMENT':
      return { count: state.count + 1 };
    
    case 'DECREMENT':
      return { count: state.count - 1 };
    
    case 'RESET':
      return { count: 0 };
    
    case 'SET_VALUE':
      return { count: action.payload };
    
    default:
      return state;
  }
}
```

### 3. Utiliser dans un composant

```tsx
function Counter() {
  const [state, dispatch] = useReducer(counterReducer, initialState);

  return (
    <div>
      <p>Compteur : {state.count}</p>
      
      <button onClick={() => dispatch({ type: 'INCREMENT' })}>
        +1
      </button>
      
      <button onClick={() => dispatch({ type: 'DECREMENT' })}>
        -1
      </button>
      
      <button onClick={() => dispatch({ type: 'RESET' })}>
        Reset
      </button>
      
      <button onClick={() => dispatch({ type: 'SET_VALUE', payload: 10 })}>
        Set to 10
      </button>
    </div>
  );
}
```

## Exemple complexe : Gestion d'un panier

### 1. Définir les types

```tsx
interface Product {
  id: number;
  name: string;
  price: number;
}

interface CartItem {
  product: Product;
  quantity: number;
}

interface CartState {
  items: CartItem[];
  total: number;
  isLoading: boolean;
}

type CartAction = 
  | { type: 'ADD_ITEM'; payload: Product }
  | { type: 'REMOVE_ITEM'; payload: number } // product id
  | { type: 'UPDATE_QUANTITY'; payload: { id: number; quantity: number } }
  | { type: 'CLEAR_CART' }
  | { type: 'SET_LOADING'; payload: boolean };
```

### 2. État initial et helpers

```tsx
const initialCartState: CartState = {
  items: [],
  total: 0,
  isLoading: false
};

// Fonction helper pour calculer le total
function calculateTotal(items: CartItem[]): number {
  return items.reduce((total, item) => total + (item.product.price * item.quantity), 0);
}
```

### 3. Reducer complexe

```tsx
function cartReducer(state: CartState, action: CartAction): CartState {
  switch (action.type) {
    case 'ADD_ITEM': {
      const existingItem = state.items.find(item => item.product.id === action.payload.id);
      
      let newItems: CartItem[];
      
      if (existingItem) {
        // Augmenter la quantité si le produit existe déjà
        newItems = state.items.map(item =>
          item.product.id === action.payload.id
            ? { ...item, quantity: item.quantity + 1 }
            : item
        );
      } else {
        // Ajouter un nouveau produit
        newItems = [...state.items, { product: action.payload, quantity: 1 }];
      }
      
      return {
        ...state,
        items: newItems,
        total: calculateTotal(newItems)
      };
    }
    
    case 'REMOVE_ITEM': {
      const newItems = state.items.filter(item => item.product.id !== action.payload);
      return {
        ...state,
        items: newItems,
        total: calculateTotal(newItems)
      };
    }
    
    case 'UPDATE_QUANTITY': {
      const newItems = state.items.map(item =>
        item.product.id === action.payload.id
          ? { ...item, quantity: action.payload.quantity }
          : item
      ).filter(item => item.quantity > 0); // Supprimer si quantité = 0
      
      return {
        ...state,
        items: newItems,
        total: calculateTotal(newItems)
      };
    }
    
    case 'CLEAR_CART':
      return {
        ...state,
        items: [],
        total: 0
      };
    
    case 'SET_LOADING':
      return {
        ...state,
        isLoading: action.payload
      };
    
    default:
      return state;
  }
}
```

### 4. Composant utilisant le panier

```tsx
function ShoppingCart() {
  const [cartState, dispatch] = useReducer(cartReducer, initialCartState);

  const addProduct = (product: Product) => {
    dispatch({ type: 'ADD_ITEM', payload: product });
  };

  const removeProduct = (productId: number) => {
    dispatch({ type: 'REMOVE_ITEM', payload: productId });
  };

  const updateQuantity = (productId: number, quantity: number) => {
    dispatch({ type: 'UPDATE_QUANTITY', payload: { id: productId, quantity } });
  };

  const clearCart = () => {
    dispatch({ type: 'CLEAR_CART' });
  };

  if (cartState.isLoading) {
    return <div>Chargement...</div>;
  }

  return (
    <div>
      <h2>Panier ({cartState.items.length} articles)</h2>
      
      {cartState.items.map(item => (
        <div key={item.product.id} className="cart-item">
          <h3>{item.product.name}</h3>
          <p>Prix : {item.product.price}€</p>
          <p>Quantité : {item.quantity}</p>
          
          <button onClick={() => updateQuantity(item.product.id, item.quantity + 1)}>
            +
          </button>
          <button onClick={() => updateQuantity(item.product.id, item.quantity - 1)}>
            -
          </button>
          <button onClick={() => removeProduct(item.product.id)}>
            Supprimer
          </button>
        </div>
      ))}
      
      <div>
        <strong>Total : {cartState.total.toFixed(2)}€</strong>
      </div>
      
      <button onClick={clearCart}>Vider le panier</button>
    </div>
  );
}
```

## useReducer avec Context API

### Combiner useReducer et Context pour un état global

```tsx
// CartContext.tsx
import { createContext, useContext, useReducer } from 'react';

const CartContext = createContext<{
  state: CartState;
  dispatch: React.Dispatch<CartAction>;
} | undefined>(undefined);

function CartProvider({ children }: { children: React.ReactNode }) {
  const [state, dispatch] = useReducer(cartReducer, initialCartState);

  return (
    <CartContext.Provider value={{ state, dispatch }}>
      {children}
    </CartContext.Provider>
  );
}

function useCart() {
  const context = useContext(CartContext);
  if (!context) {
    throw new Error('useCart doit être utilisé dans CartProvider');
  }
  return context;
}

// Actions helpers
function useCartActions() {
  const { dispatch } = useCart();

  return {
    addItem: (product: Product) => dispatch({ type: 'ADD_ITEM', payload: product }),
    removeItem: (id: number) => dispatch({ type: 'REMOVE_ITEM', payload: id }),
    updateQuantity: (id: number, quantity: number) => 
      dispatch({ type: 'UPDATE_QUANTITY', payload: { id, quantity } }),
    clearCart: () => dispatch({ type: 'CLEAR_CART' })
  };
}

export { CartProvider, useCart, useCartActions };
```

## useState vs useReducer

### ✅ Utiliser useState quand :
- État simple (string, number, boolean)
- Peu de logique de mise à jour
- Composants isolés

### ✅ Utiliser useReducer quand :
- État complexe avec plusieurs propriétés
- Logique de mise à jour complexe
- Plusieurs actions possibles
- Besoin de prévisibilité (tests plus faciles)
- État partagé entre composants (avec Context)

---

## Bonnes pratiques

### 1. Typer les actions avec union types
```tsx
type Action = 
  | { type: 'LOADING' }
  | { type: 'SUCCESS'; payload: Data }
  | { type: 'ERROR'; payload: string };
```

### 2. Utiliser des constantes pour les types d'actions
```tsx
const ACTIONS = {
  LOADING: 'LOADING',
  SUCCESS: 'SUCCESS',
  ERROR: 'ERROR'
} as const;
```

### 3. Séparer la logique complexe
```tsx
// Extraire les fonctions helpers
const addItemToCart = (items: CartItem[], product: Product): CartItem[] => {
  // logique...
};

function cartReducer(state: CartState, action: CartAction): CartState {
  switch (action.type) {
    case 'ADD_ITEM':
      return {
        ...state,
        items: addItemToCart(state.items, action.payload)
      };
  }
}
```

### 4. Tester les reducers
```tsx
// Les reducers sont des fonctions pures, faciles à tester
test('should add item to cart', () => {
  const initialState = { items: [], total: 0 };
  const product = { id: 1, name: 'Test', price: 10 };
  
  const newState = cartReducer(initialState, { type: 'ADD_ITEM', payload: product });
  
  expect(newState.items).toHaveLength(1);
  expect(newState.total).toBe(10);
});
```
